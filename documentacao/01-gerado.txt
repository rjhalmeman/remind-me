exports.login = async function(req, res) { const { email, senha } = req.body;

try {
    const result = await db.query(
        'SELECT * FROM pessoa WHERE email_pessoa = $1 AND senha_pessoa = $2',
        [email, senha]
    );

    if (result.rows.length > 0) {
        const usuario = result.rows[0];
        // Cria um cookie simples (para fins didáticos)
        res.cookie('usuario_logado', usuario.id_pessoa, { maxAge: 900000, httpOnly: false });
        res.cookie('nome_usuario', usuario.nome_pessoa, { maxAge: 900000, httpOnly: false });
        res.json({ sucesso: true, usuario: usuario });
    } else {
        res.status(401).json({ sucesso: false, mensagem: 'Credenciais inválidas' });
    }
} catch (erro) {
    res.status(500).json({ erro: erro.message });
}
};

exports.logout = function(req, res) { res.clearCookie('usuario_logado'); res.clearCookie('nome_usuario'); res.json({ sucesso: true }); };

exports.criar = async function(req, res) { const { nome, data, hora, status, id_pessoa, descricao } = req.body; try { // Inserir evento const insertEvento = await db.query( 'INSERT INTO eventos (nome_evento, data_evento, hora_evento, status, id_pessoa) VALUES ($1, $2, $3, $4, $5) RETURNING id_evento', [nome, data, hora, status, id_pessoa] );

    // Inserir descrição

    }

} catch (err) {
    res.status(500).json({ erro: err.message });
}
};

// frontend/menu/menu.js window.onload = function() { carregarUsuario(); carregarEventosMenu(); };

function carregarUsuario() { // Ler cookie 'nome_usuario' de forma simples var match = document.cookie.match(new RegExp('(^| )nome_usuario=([^;]+)')); if (match) { // Decodificar URI caso tenha espaços/acentos document.getElementById('nomeUsuario').innerText = decodeURIComponent(match[2]).replace(/%20/g, ' '); } else { window.location.href = '../login/login.html'; } }

function fazerLogout() { fetch('/api/auth/logout', { method: 'POST' }) .then(function() { window.location.href = '../login/login.html'; }); }

function carregarEventosMenu() { fetch('/api/eventos/menu') .then(function(res) { return res.json(); }) .then(function(eventos) { var html = ''; if (eventos.length === 0) { html = '<p class="text-center">Nenhum evento pendente.</p>'; } else { for (var i = 0; i < eventos.length; i++) { var e = eventos[i]; // Formatar data var dataFormatada = new Date(e.data_evento).toLocaleDateString('pt-BR');

            // Definir classe de cor baseada no ID do status
            var classeStatus = 'status-' + e.status;

            html += '<div class="card-evento ' + classeStatus + '">';
            html += '<div>';
            html += '<h3>' + e.nome_evento + '</h3>';
            html += '<p>Data: ' + dataFormatada + ' às ' + e.hora_evento + '</p>';
            html += '<p><i>' + (e.descricao_evento || 'Sem descrição') + '</i></p>';
            html += '<b>Status: ' + e.descricao_status + '</b>';
            html += '</div>';
            
            // Botões de ação rápida (Mudar status)
            html += '<div>';
            if (e.status == 1) { // Se Agendado -> Botão Iniciar
                html += '<button onclick="mudarStatus('+e.id_evento+', 2)">Iniciar</button> ';
                html += '<button class="btn-danger" onclick="mudarStatus('+e.id_evento+', 3)">Cancelar</button>';
            } else if (e.status == 2) { // Se Em andamento -> Botão Concluir
                html += '<button onclick="mudarStatus('+e.id_evento+', 4)">Concluir</button>';
            }
            html += '</div>';
            html += '</div>';
        }
    }
    document.getElementById('listaEventos').innerHTML = html;
});
}

function mudarStatus(idEvento, novoStatus) { fetch('/api/eventos/status/' + idEvento, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ novoStatus: novoStatus }) }) .then(function() { carregarEventosMenu(); }); }

function carregarSelectStatus() { fetch('/api/status') .then(function(res) { return res.json(); }) .then(function(lista) { var select = document.getElementById('status_evento'); select.innerHTML = ''; for (var i = 0; i < lista.length; i++) { var opt = document.createElement('option'); opt.value = lista[i].id_status; opt.innerText = lista[i].descricao_status; select.appendChild(opt); } }); }

function listarEventos() { fetch('/api/eventos') .then(function(res) { return res.json(); }) .then(function(lista) { var tbody = document.getElementById('tabelaEventos'); tbody.innerHTML = ''; for(var i=0; i < lista.length; i++) { var e = lista[i]; var tr = '<tr>' + '<td>' + e.id_evento + '</td>' + '<td>' + e.nome_evento + '</td>' + '<td>' + new Date(e.data_evento).toLocaleDateString() + '</td>' + '<td>' + '<button onclick="deletarEvento('+e.id_evento+')" class="btn-danger">Excluir</button>' + '</td>' + '</tr>'; tbody.innerHTML += tr; } }); }

function salvarEvento() { var nome = document.getElementById('nome_evento').value; var data = document.getElementById('data_evento').value; var hora = document.getElementById('hora_evento').value; var desc = document.getElementById('descricao_evento').value; var status = document.getElementById('status_evento').value; var pessoa = document.getElementById('id_pessoa').value;

var dados = {
    nome: nome, data: data, hora: hora, descricao: desc, status: status, id_pessoa: pessoa
};

fetch('/api/eventos', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(dados)
})
.then(function() {
    alert('Salvo com sucesso!');
    limparForm();
    listarEventos();
})
.catch(function(err) { alert('Erro ao salvar'); });
}

function deletarEvento(id) { if(confirm('Tem certeza?')) { fetch('/api/eventos/' + id, { method: 'DELETE' }) .then(function() { listarEventos(); }); } }
